    async function syncFromSupabase(classCode) {

          supabaseClient.from('teachers').select('*'),

          supabaseClient.from('classes').select('*'),

          classCode ? supabaseClient.from('students').select('*').eq('class_code', classCode) : supabaseClient.from('students').select('*'),

          classCode ? supabaseClient.from('assignments').select('*').eq('class_code', classCode) : supabaseClient.from('assignments').select('*')

        ]);


        if (!tRes.error && tRes.data && tRes.data.length > 0) db.teachers = tRes.data;

        const adminEmail = CONFIG.ADMIN_TEACHER_EMAIL.toLowerCase();

        if (!db.teachers.some(t => t.email === adminEmail)) {

          db.teachers.unshift({ email: adminEmail, name: "Liam McDermott", is_admin: true });

        }

        if (!cRes.error && Array.isArray(cRes.data)) db.classes = cRes.data;

        if (!sRes.error && Array.isArray(sRes.data)) {

          db.students = classCode

            ? db.students.filter(s => s.class_code !== classCode).concat(sRes.data)

            : sRes.data;

      }

    }

    function currentUnitMorphemes() {

      const unit = CURRICULUM[currentSetKey][currentUnitIndex];

      const cleaned = unit.id.replace(/\(.*?\)/g, '');

      return cleaned

        .split(/[/,]/)

        .map(part => part.replace(/[^a-zA-Z]/g, '').toLowerCase())

        .filter(Boolean)

        .sort((a, b) => b.length - a.length);

    }


    function extractBaseTileFromExample(exampleWord, unit) {

      const parts = currentUnitMorphemes();

      for (const p of parts) {

        if (unit.type === "Prefix" && exampleWord.startsWith(p) && exampleWord.length > p.length + 1) {

          return { affix: p + "-", base: exampleWord.slice(p.length), isPrefix: true };

        }

        if (unit.type === "Suffix" && exampleWord.endsWith(p) && exampleWord.length > p.length + 1) {

          return { affix: "-" + p, base: exampleWord.slice(0, -p.length), isPrefix: false };

        }

      }

      return { affix: unit.id, base: exampleWord, isPrefix: true };

    }

    function setupSorterGame(unit) {

      const extras = EXTRA_NON_EXAMPLES[unit.id] || [];

      const tricksters = [...new Set([...unit.nonExamples, ...extras])].slice(0, 5);

      const reals = unit.examples.slice(0, 5);


      sorterQueue = [

        ...tricksters.map(w => ({ word: w, isReal: false })),

        ...reals.map(w => ({ word: w, isReal: true }))

      ].sort(() => Math.random() - 0.5);


      document.getElementById('nonExamplesContainer').innerHTML = '';

      nextSorterWord();

    }

    /* DICTIONARY VERIFICATION FOR BONUS TYPING */

    async function isWordInDictionary(word) {

      if (dictionaryCache.has(word)) return dictionaryCache.get(word);

      try {

        const res = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`);

        if (!res.ok) return false;

        const data = await res.json();

        const valid = Array.isArray(data) && data.some(e => e.word && e.word.toLowerCase() === word);

        dictionaryCache.set(word, valid);

        return valid;

      } catch (e) {

        return false;

      }

    }


    async function submitWord() {

      const input = document.getElementById('wordInput');

      const btn = document.getElementById('checkWordBtn');

      const word = input.value.trim().toLowerCase().replace(/[^a-z]/g, '');

      if (!word) return;


      const unit = CURRICULUM[currentSetKey][currentUnitIndex];

      const extras = EXTRA_NON_EXAMPLES[unit.id] || [];


      if (unit.nonExamples.includes(word) || extras.includes(word)) {

        addWordChip(word, true);

        speakOwly(`Aha! ${word} is a trickster word!`);

        input.value = '';

        return;

      }


      btn.disabled = true;

      btn.textContent = "Checking...";

      const inDict = unit.examples.includes(word) || await isWordInDictionary(word);

      btn.disabled = false;

      btn.textContent = "Check Word";


      if (!inDict) {

        speakOwly(`Hmm, I couldn't find ${word} in the dictionary. Have another go!`);

        input.value = '';

        return;

      }


      const parts = currentUnitMorphemes();

      const matches = unit.type === "Prefix"

        ? parts.some(p => word.startsWith(p) && word.length >= p.length + 2)

        : unit.type === "Suffix"

          ? parts.some(p => word.endsWith(p) && word.length >= p.length + 2)

          : parts.some(p => word.includes(p) && word.length >= p.length + 1);


      if (matches) {

        await awardDiscoveredWord(word, `Splendid! ${word} is in your word bank!`);

      } else {

        speakOwly(`Good try, ${word} is a real word, but it doesn't use ${unit.id}.`);

      }

      input.value = '';

    }

    function launchGradedTest(scope, questionCount, title, assignmentId) {

      let pool = [];

      if (scope === 'set1' || scope === 'set2' || scope === 'set3') {

        pool = CURRICULUM[scope];

      } else {

        const [sKey, uId] = scope.split('_');

        const found = (CURRICULUM[sKey] || []).find(u => u.id === uId);

        pool = found ? [found] : CURRICULUM.set1;

      }


      const questions = [];

      for (let i = 0; i < questionCount; i++) {

        const unit = pool[i % pool.length];

        const mode = i % 3;

        if (mode === 0) {

          const wrong = CURRICULUM.set1.filter(u => u.meaning !== unit.meaning).sort(() => Math.random() - 0.5).slice(0, 2).map(u => u.meaning);

          questions.push({

            prompt: `What does "${unit.id}" mean?`,

            correct: unit.meaning,

            options: [unit.meaning, ...wrong].sort(() => Math.random() - 0.5)

          });

        } else if (mode === 1) {

          const correct = unit.examples[i % unit.examples.length];

          const w1 = unit.nonExamples[0] || "uncle";

          questions.push({

            prompt: `Which word uses "${unit.id}" (${unit.meaning})?`,

            correct,

            options: [correct, w1, "elephant"].sort(() => Math.random() - 0.5)

          });

        } else {

          const trick = unit.nonExamples[0] || "uncle";

          questions.push({

            prompt: `Which word is a trickster for "${unit.id}"?`,

            correct: trick,

            options: [trick, unit.examples[0], unit.examples[1]].sort(() => Math.random() - 0.5)

          });

        }

      }


      activeQuiz = { questions, currentIndex: 0, score: 0, assignmentId, title };

      document.getElementById('lessonWorkspace').style.display = 'none';

      document.getElementById('quizWorkspace').style.display = 'flex';

      renderQuizQuestion();

    }

    }


    function downloadCSV(rows, filename) {

      const csv = "data:text/csv;charset=utf-8," + rows.map(r => r.join(",")).join("\n");

      const a = document.createElement("a");

      a.href = encodeURI(csv);

      a.download = filename;

      document.body.appendChild(a);

      a.click();

      document.body.removeChild(a);
